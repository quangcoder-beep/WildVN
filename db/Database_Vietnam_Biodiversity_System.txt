CREATE TYPE media_type AS ENUM ('IMAGE', 'AUDIO', 'VIDEO');

CREATE TYPE user_role AS ENUM ('USER', 'ADMIN', 'MODERATOR');

-- Users Table
CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    username VARCHAR(100) NOT NULL,
    email VARCHAR(255) NOT NULL,
    password_hash VARCHAR(255) NOT NULL, -- Assumes Argon2id handled in backend
    role user_role DEFAULT 'USER' NOT NULL,
    xp INT DEFAULT 0 NOT NULL,
    level INT DEFAULT 1 NOT NULL,
    
    -- Audit fields
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP NOT NULL,
    deleted_at TIMESTAMP WITH TIME ZONE,
    
    CONSTRAINT chk_xp_positive CHECK (xp >= 0),
    CONSTRAINT chk_level_positive CHECK (level >= 1),
    CONSTRAINT chk_username_format CHECK (username ~ '^[a-zA-Z0-9_]{3,30}$'),
    CONSTRAINT chk_email_format CHECK (email ~* '^[A-Za-z0-9._%-]+@[A-Za-z0-9.-]+[.][A-Za-z]+$')
);


-- Locations
CREATE TABLE locations (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    name VARCHAR(255) NOT NULL,
    type location_type NOT NULL, 
    geom GEOMETRY(Geometry, 4326) NOT NULL, 
    
    created_by UUID REFERENCES users(id) ON DELETE SET NULL,
    updated_by UUID REFERENCES users(id) ON DELETE SET NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP NOT NULL,
    deleted_at TIMESTAMP WITH TIME ZONE,
    
    CONSTRAINT chk_location_name_not_empty CHECK (length(trim(name)) > 0),
    CONSTRAINT chk_valid_geometry CHECK (GeometryType(geom) IN ('POINT', 'POLYGON', 'MULTIPOLYGON'))
);

-- Taxonomy
CREATE TABLE taxonomy (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    parent_id UUID REFERENCES taxonomy(id) ON DELETE RESTRICT,
    rank taxonomy_rank NOT NULL,
    scientific_name VARCHAR(255) NOT NULL,
    name_vn VARCHAR(255),
    
    created_by UUID REFERENCES users(id) ON DELETE SET NULL,
    updated_by UUID REFERENCES users(id) ON DELETE SET NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP NOT NULL,
    deleted_at TIMESTAMP WITH TIME ZONE,
    
    CONSTRAINT chk_taxonomy_parent CHECK (id != parent_id),
    CONSTRAINT chk_taxonomy_scientific_name_not_empty CHECK (length(trim(scientific_name)) > 0)
);

-- Species
CREATE TABLE species (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    taxonomy_id UUID NOT NULL REFERENCES taxonomy(id) ON DELETE RESTRICT,
    slug VARCHAR(255) NOT NULL,
    scientific_name VARCHAR(255) NOT NULL,
    name_vn VARCHAR(255) NOT NULL,
    conservation_status conservation_status NOT NULL,
    identification_desc TEXT,
    biology_desc TEXT,
    threats_desc TEXT,
    conservation_desc TEXT,
    
    -- Optimized Full-text search Vector
    search_vector TSVECTOR GENERATED ALWAYS AS (
        setweight(to_tsvector('simple', coalesce(scientific_name, '')), 'A') ||
        setweight(to_tsvector('simple', coalesce(name_vn, '')), 'A') ||
        setweight(to_tsvector('simple', coalesce(identification_desc, '')), 'B') ||
        setweight(to_tsvector('simple', coalesce(biology_desc, '')), 'C') ||
        setweight(to_tsvector('simple', coalesce(threats_desc, '')), 'C') ||
        setweight(to_tsvector('simple', coalesce(conservation_desc, '')), 'C')
    ) STORED,

    created_by UUID REFERENCES users(id) ON DELETE SET NULL,
    updated_by UUID REFERENCES users(id) ON DELETE SET NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP NOT NULL,
    deleted_at TIMESTAMP WITH TIME ZONE,
    
    CONSTRAINT chk_species_slug_format CHECK (slug ~ '^[a-z0-9]+(?:-[a-z0-9]+)*$'),
    CONSTRAINT chk_species_name_vn_not_empty CHECK (length(trim(name_vn)) > 0),
    CONSTRAINT chk_species_scientific_name_not_empty CHECK (length(trim(scientific_name)) > 0)
);

-- Auto-update `updated_at`
CREATE OR REPLACE FUNCTION trigger_set_timestamp() RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = CURRENT_TIMESTAMP;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Validate taxonomy hierarchy ranks (Parent Rank > Child Rank)
CREATE OR REPLACE FUNCTION validate_taxonomy_hierarchy() RETURNS TRIGGER AS $$
DECLARE
    parent_rank taxonomy_rank;
    rank_value INT;
    parent_rank_value INT;
BEGIN
    IF NEW.parent_id IS NOT NULL THEN
        SELECT rank INTO parent_rank FROM taxonomy WHERE id = NEW.parent_id;
        
        rank_value := CASE NEW.rank
            WHEN 'KINGDOM' THEN 1 WHEN 'PHYLUM' THEN 2 WHEN 'CLASS' THEN 3
            WHEN 'ORDER' THEN 4 WHEN 'FAMILY' THEN 5 WHEN 'GENUS' THEN 6 END;
        
        parent_rank_value := CASE parent_rank
            WHEN 'KINGDOM' THEN 1 WHEN 'PHYLUM' THEN 2 WHEN 'CLASS' THEN 3
            WHEN 'ORDER' THEN 4 WHEN 'FAMILY' THEN 5 WHEN 'GENUS' THEN 6 END;
        
        IF rank_value <= parent_rank_value THEN
            RAISE EXCEPTION 'Invalid taxonomy hierarchy: Parent rank (%) must be strictly higher than Child rank (%)', parent_rank, NEW.rank;
        END IF;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER validate_taxonomy_parent_rank BEFORE INSERT OR UPDATE ON taxonomy
FOR EACH ROW EXECUTE FUNCTION validate_taxonomy_hierarchy();

-- 7. CACHING & READ OPTIMIZATION (MATERIALIZED VIEWS)
-- =======================================================================================
-- Note: Refresh these concurrently via CRON: `REFRESH MATERIALIZED VIEW CONCURRENTLY mv_leaderboard;`
CREATE MATERIALIZED VIEW mv_leaderboard AS
SELECT id, username, xp, level
FROM users
WHERE deleted_at IS NULL
ORDER BY xp DESC, created_at DESC;
CREATE UNIQUE INDEX idx_mv_leaderboard_id ON mv_leaderboard(id);

CREATE MATERIALIZED VIEW mv_species_summary AS
SELECT s.id, s.scientific_name, s.name_vn, t.rank, t.name_vn as taxonomy_name_vn, s.conservation_status,
       COUNT(up.user_id) as total_collectors
FROM species s
JOIN taxonomy t ON s.taxonomy_id = t.id
LEFT JOIN user_pokedex up ON s.id = up.species_id
WHERE s.deleted_at IS NULL
GROUP BY s.id, s.scientific_name, s.name_vn, t.rank, t.name_vn, s.conservation_status;
CREATE UNIQUE INDEX idx_mv_species_summary_id ON mv_species_summary(id);

-- =======================================================================================
-- 8. OPERATIONAL READINESS & MIGRATION STRATEGY
-- =======================================================================================
-- Migration (Zero Downtime):
--  - Always run `CREATE INDEX CONCURRENTLY` in production.
--  - Apply constraints using `NOT VALID` followed by `ALTER TABLE ... VALIDATE CONSTRAINT`.
-- Backup (PITR):
--  - Enable WAL archiving (`archive_mode = on`). Base backups daily via `pg_basebackup`. 
--  - Ensure S3/Blob storage configuration guarantees 30-day Point-in-Time-Recovery window.
-- Monitoring:
--  - `pg_stat_statements` enabled. Monitor Buffer Cache Hit Ratio (Goal > 99%).
--  - Use infrastructure-level logging (Nginx/API Gateway/ELK) for abuse detection.
